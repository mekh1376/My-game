function drawWheel() { const numSegments = items.length; const arc = (2 * Math.PI) / numSegments; const centerX = 130; const centerY = 130; const radius = 130; for (let i = 0; i < numSegments; i++) { const angle = i * arc; ctx.beginPath(); ctx.fillStyle = colors[i]; ctx.moveTo(centerX, centerY); ctx.arc(centerX, centerY, radius, angle, angle + arc); ctx.fill(); // خطوط جداکننده قاچ‌ها ctx.beginPath(); ctx.strokeStyle = 'rgba(251, 191, 36, 0.4)'; ctx.lineWidth = 2; ctx.moveTo(centerX, centerY); ctx.lineTo(centerX + Math.cos(angle) * radius, centerY + Math.sin(angle) * radius); ctx.stroke(); // تنظیم موقعیت متن در مرکز هر بخش ctx.save(); ctx.translate(centerX, centerY); // چرخش به مرکز زاویه قاچ ctx.rotate(angle + arc / 2); ctx.fillStyle = '#ffffff'; ctx.font = 'bold 12px Tahoma, Arial'; ctx.textAlign = 'center'; // قرار دادن نقطه ثقل متن در وسط کلمه ctx.textBaseline = 'middle'; // ترازبندی عمودی دقیق در مرکز ctx.shadowColor = 'rgba(0, 0, 0, 0.9)'; ctx.shadowBlur = 4; ctx.shadowOffsetX = 1; ctx.shadowOffsetY = 1; // عدد 75 فاصله شعاعی از مرکز دایره است (دقیقاً وسط شعاع 130 پیکسلی) ctx.fillText(items[i], 75, 0); ctx.restore(); } // دایره کوچک وسط گردونه ctx.beginPath(); ctx.fillStyle = '#fbbf24'; ctx.arc(centerX, centerY, 18, 0, 2 * Math.PI); ctx.fill(); ctx.lineWidth = 3; ctx.strokeStyle = '#ffffff'; ctx.stroke(); }
